cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(TitanFileFilter VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Build with Visual Studio C++ tools on Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TRACE_ASAN "Instrument all user-mode targets with AddressSanitizer" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(TRACE_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
add_library(FileFilter SDK/FileFilter.cpp)
target_link_libraries(FileFilter PUBLIC fltlib)
add_executable(TitanFileFilter WIN32 App/Main.cpp)
target_link_libraries(TitanFileFilter PRIVATE FileFilter user32 gdi32 comctl32 comdlg32 shell32 ole32)
target_link_options(TitanFileFilter PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
add_executable(FileFilterCLI App/Console.cpp)
target_link_libraries(FileFilterCLI PRIVATE FileFilter)
add_executable(FileFilterExample Examples/protect.cpp)
target_link_libraries(FileFilterExample PRIVATE FileFilter)
add_executable(FileFilterTests Tests/ProtocolTests.cpp)
target_link_libraries(FileFilterTests PRIVATE FileFilter TraceSetupCore ole32)
enable_testing()
add_test(NAME OfflineProtocolScopeAndPackageGuards COMMAND FileFilterTests)

set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Opaque signed TitanFileFilter INF/CAT/SYS package")
set(PACKAGE_RESOURCE "")
if(DRIVER_PACKAGE_DIRECTORY)
 enable_language(RC)
 foreach(name TitanFileFilter.inf TitanFileFilter.cat TitanFileFilter.sys)
  if(NOT EXISTS "${DRIVER_PACKAGE_DIRECTORY}/${name}")
   message(FATAL_ERROR "Incomplete opaque driver package: ${name}")
  endif()
 endforeach()
 set(PACKAGE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/DriverPackage.rc")
 file(WRITE "${PACKAGE_RESOURCE}" "101 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanFileFilter.inf\"\n102 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanFileFilter.cat\"\n103 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanFileFilter.sys\"\n")
endif()
add_library(TraceSetupCore Installer/Setup.cpp)
target_link_libraries(TraceSetupCore PUBLIC setupapi newdev wintrust crypt32 advapi32 ole32 fltlib)
add_executable(FileFilterSetup WIN32 Installer/Main.cpp ${PACKAGE_RESOURCE})
add_executable(FileFilterSetupConsole Installer/Console.cpp ${PACKAGE_RESOURCE})
foreach(target FileFilterSetup FileFilterSetupConsole)
 target_link_libraries(${target} PRIVATE TraceSetupCore user32 gdi32)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()

foreach(target TitanFileFilter FileFilterCLI FileFilterExample FileFilterTests FileFilterSetup FileFilterSetupConsole)
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()

add_test(NAME OfflineConsolePreview COMMAND FileFilterCLI --preview)
